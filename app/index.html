import gradio as gr import pandas as pd from sklearn.compose import ColumnTransformer from sklearn.ensemble import RandomForestClassifier from sklearn.impute import SimpleImputer from sklearn.pipeline import Pipeline from sklearn.preprocessing import OrdinalEncoder, StandardScaler def train_from_csv(): df = pd.read_csv("drug.csv") X = df.drop("Drug", axis=1).values y = df["Drug"].values transform = ColumnTransformer( [ ("encoder", OrdinalEncoder(), [1, 2, 3]), ("num_imputer", SimpleImputer(strategy="median"), [0, 4]), ("num_scaler", StandardScaler(), [0, 4]), ] ) model = Pipeline( steps=[ ("preprocessing", transform), ("model", RandomForestClassifier(n_estimators=100, random_state=125)), ] ) model.fit(X, y) return model try: import micropip await micropip.install("skops") import skops.io as sio path = "drug_pipeline.skops" pipe = sio.load(path, trusted=sio.get_untrusted_types(file=path)) pipe.predict([[30, "M", "HIGH", "NORMAL", 20.0]]) except Exception: pipe = train_from_csv() def predict_drug(age, sex, blood_pressure, cholesterol, na_to_k): features = [age, sex, blood_pressure, cholesterol, na_to_k] return "Predicted Drug: " + str(pipe.predict([features])[0]) inputs = [ gr.Slider(15, 74, step=1, label="Age"), gr.Radio(["M", "F"], label="Sex"), gr.Radio(["HIGH", "LOW", "NORMAL"], label="Blood Pressure"), gr.Radio(["HIGH", "NORMAL"], label="Cholesterol"), gr.Slider(6.2, 38.2, step=0.1, label="Na_to_K"), ] demo = gr.Interface( fn=predict_drug, inputs=inputs, outputs="text", title="Drug Classification", ) demo.launch() scikit-learn pandas
